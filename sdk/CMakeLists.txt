# The sound engine SDK for the PC (docs/README.md): every engine as its own library, a generic
# contract test suite that runs on each of them, the renderer and the host shell. It needs only
# dsp/ (the engines), tools/harness/ (the scripted performances and the key-stream replay) and
# third_party/doctest: no OS, no device. Built as part of an including project, or on its own:  cmake -S sdk -B build/sdk && cmake --build build/sdk && ctest --test-dir build/sdk
#
#   awva_engine_<name>       one engine (dsp/include/awva/engines/<name>.h): its EngineEntry,
#                              awva_engine_<name>()
#   awva_effect_fx           the master effects: awva::sdk::fx_effect()
#   awva_engines             every engine, by name: awva::sdk::engines(), find_engine()
#   awva_performances        the scripted performances of tools/harness, by name
#   awva_sdk                 the MIDI/MPE mapping, the MIDI file reader, the players, the shell
#   awva_render              the renderer (sdk/tools/awva_render.cpp)
#   render_custom              the renderer with one engine and performance compiled with extra
#                              flags (AWVA_RENDER_*)
#   tests: contract.<name>.*   the contract suite on each engine (tests/contract*.cpp, docs/conformance.md;
#                              contract.<name>.soak: the label soak, not run by default)
#          sdk.*               the shared helpers, the MIDI mapping and files, the shell
#          awva_render.*     the renderer's log names its engine, preset and input; bad inputs fail;
#                            the fitting tools' options (--params, the controls round trip)
#          golden.*          every engine's renders against their hashes (tests/goldens/renders.txt)

# AWVA_HARNESS_DIR, AWVA_DOCTEST_DIR: this repository's unless given (-D, or by the including
# project). Not cached: a build directory configured from another checkout or before a move
# can't keep a stale path. One given that doesn't exist stops the configure.
foreach(_v HARNESS_DIR:tools/harness DOCTEST_DIR:third_party/doctest)
  string(REPLACE ":" ";" _p "${_v}")
  list(GET _p 0 _name)
  list(GET _p 1 _rel)
  if(NOT AWVA_${_name})
    set(AWVA_${_name} "${CMAKE_CURRENT_SOURCE_DIR}/../${_rel}")
  elseif(NOT EXISTS "${AWVA_${_name}}")
    message(FATAL_ERROR "AWVA_${_name}=${AWVA_${_name}} doesn't exist (a stale cache? configure the build directory again, or drop -DAWVA_${_name})")
  endif()
endforeach()

# --- the scripted performances (tools/harness/performance.h, perf_<name>.h) ----------------------------
function(awva_performance_source out name header)
  string(MAKE_C_IDENTIFIER "${name}" id)
  file(CONFIGURE OUTPUT "${out}" @ONLY CONTENT "// generated by sdk/CMakeLists.txt: the scripted performance \"${name}\" (${header})
#include <math.h>
#include <stdint.h>
#include <stdio.h>
#include <string.h>

#include \"awva_sdk/performance.h\"

namespace perf_${id} {
#include \"${header}\"
}

static void perf_block(int b, int nv, awva::dsp::VoiceControls* vc, awva::dsp::GlobalControls& g) {
	perf_${id}::Ctl v[NVOICES_MAX];
	float gl[8];
	perf_${id}::block_controls(b, v, nv, gl);
	for (int i = 0; i < nv; i++) {
		vc[i].note = v[i].note;
		vc[i].pres = v[i].pres;
		vc[i].vpres = v[i].vpres;
		vc[i].but_x = v[i].but_x;
		vc[i].but_y = v[i].but_y;
	}
	g = {gl[0], gl[1], gl[2], gl[3], gl[4], gl[5], gl[6], gl[7]};
}

extern const awva::sdk::ScriptedPerformance awva_perf_${id};
const awva::sdk::ScriptedPerformance awva_perf_${id} = {\"${name}\", \"${header}\", SAMPLERATE, NVOICES_MAX, perf_block};
")
endfunction()

file(GLOB _perf_headers "${AWVA_HARNESS_DIR}/perf_*.h")

list(PREPEND _perf_headers "${AWVA_HARNESS_DIR}/performance.h")
set(_perf_srcs "")
set(_perf_decls "")
set(_perf_list "")
set(AWVA_PERFORMANCES "")
foreach(h IN LISTS _perf_headers)
  get_filename_component(hn "${h}" NAME_WE)
  if(hn STREQUAL "performance")
    set(name default)
  else()
    string(REGEX REPLACE "^perf_" "" name "${hn}")
  endif()
  string(MAKE_C_IDENTIFIER "${name}" id)
  awva_performance_source("${_gen}/perf_${id}.cpp" "${name}" "${h}")
  list(APPEND _perf_srcs "${_gen}/perf_${id}.cpp")
  list(APPEND AWVA_PERFORMANCES ${name})
  string(APPEND _perf_decls "extern const awva::sdk::ScriptedPerformance awva_perf_${id};\n")
  string(APPEND _perf_list "\t    &awva_perf_${id},\n")
endforeach()

# The golden renders (sdk/tests/goldens/renders.txt): every engine and preset on the scripted
# performances and the recorded takes, and every engine on the MIDI phrases (MPE, MIDI 2.0; also in
# mono legato), each checked against its FNV-1a 64 hash: a change of any engine's output or of the
# MIDI mapping fails here. Only a build of the SDK itself (a project that includes it may add engines
# of its own). After an intended change: tools/goldens/update.sh <build dir> (and say so).
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(_golden_dir "${CMAKE_CURRENT_SOURCE_DIR}/tests/goldens")
  set(_golden_file "${_golden_dir}/renders.txt")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_golden_file}")
  if(EXISTS "${_golden_file}")
    file(STRINGS "${_golden_file}" _golden_lines REGEX "^[^#]")
  else()
    set(_golden_lines "")
  endif()
  foreach(l IN LISTS _golden_lines)
    if(l MATCHES "^([^ ]+) ([0-9a-f]+)$")
      set("_golden_${CMAKE_MATCH_1}" "${CMAKE_MATCH_2}")
    endif()
  endforeach()
  file(GLOB _takes "${AWVA_HARNESS_DIR}/rec/*.csv")
  list(SORT _takes)
  function(_golden name)
    if(DEFINED "_golden_${name}")
      set(h "${_golden_${name}}")
    else()
      set(h missing)
    endif()
    add_test(NAME ${name} COMMAND awva_render --nv 10 --check --golden-name ${name} --expect ${h} ${ARGN})
    set_tests_properties(${name} PROPERTIES LABELS golden)
  endfunction()
  foreach(v IN LISTS AWVA_ENGINES)
    set(np "${AWVA_ENGINE_${v}_PRESETS}")
    set(presets none)
    if(np)
      math(EXPR last "${np} - 1")
      set(presets "")
      foreach(p RANGE ${last})
        list(APPEND presets ${p})
      endforeach()
    endif()
    foreach(p IN LISTS presets)
      if(p STREQUAL "none")
        set(tag "${v}")
        set(parg "")
      else()
        set(tag "${v}.p${p}")
        set(parg --preset ${p})
      endif()
      _golden(golden.${tag}.default --engine ${v} ${parg} --perf default)
      # (a released voice's last pressure velocity left positive: silent and idle 12 s after it)
      _golden(golden.${tag}.stale --engine ${v} ${parg} --perf stale --silent-after 12)
      foreach(take IN LISTS _takes)
        get_filename_component(tn "${take}" NAME_WE)
        string(REGEX REPLACE "^play_" "" tn "${tn}")
        _golden(golden.${tag}.take_${tn} --engine ${v} ${parg} --rec "${take}")
      endforeach()
    endforeach()
    if(v IN_LIST AWVA_PERFORMANCES) # its own performance, tools/harness/perf_<v>.h
      _golden(golden.${v}.perf --engine ${v} --perf ${v})
    endif()
    foreach(ph IN ITEMS mpe midi2)
      _golden(golden.${v}.${ph} --engine ${v} --ump "${_golden_dir}/${ph}.ump")
      _golden(golden.${v}.${ph}.legato --engine ${v} --ump "${_golden_dir}/${ph}.ump" --legato)
    endforeach()
  endforeach()
  _golden(golden.va.p5.fx --engine va --preset 5 --perf default --fx)
endif()
