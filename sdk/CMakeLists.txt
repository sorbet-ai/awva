# The sound engine SDK for the PC (docs/README.md): every engine as its own library, a generic
# contract test suite that runs on each of them, the renderer and the host shell. It needs only
# dsp/ (the engines), tools/harness/ (the scripted performances and the key-stream replay) and
# third_party/doctest: no OS, no device. Built as part of an including project, or on its own:  cmake -S sdk -B build/sdk && cmake --build build/sdk && ctest --test-dir build/sdk
#
#   awva_engine_<name>       one engine (dsp/include/awva/engines/<name>.h): its EngineEntry,
#                              awva_engine_<name>()
#   awva_effect_fx           the master effects: awva::sdk::fx_effect()
#   awva_engines             every engine, by name: awva::sdk::engines(), find_engine()
#   awva_performances        the scripted performances of tools/harness, by name
#   awva_sdk                 the MIDI/MPE mapping, the MIDI file reader, the players, the shell
#   awva_render              the renderer (sdk/tools/awva_render.cpp)
#   render_custom              the renderer with one engine and performance compiled with extra
#                              flags (AWVA_RENDER_*)
#   tests: contract.<name>.*   the contract suite on each engine (tests/contract.cpp)
#          sdk.*               the shared helpers, the MIDI mapping and files, the shell
#          awva_render.*     the renderer's log names its engine, preset and input; bad inputs fail
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.25)
  project(awva_sdk DESCRIPTION "awva sound engine SDK" LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../dsp" "${CMAKE_BINARY_DIR}/dsp")
  add_library(awva_host_options INTERFACE)
  target_compile_options(awva_host_options INTERFACE -Wall -Wextra)
  enable_testing()
endif()
# AWVA_HARNESS_DIR, AWVA_DOCTEST_DIR: this repository's unless given (-D, or by the including
# project). Not cached: a build directory configured from another checkout or before a move
# can't keep a stale path. One given that doesn't exist stops the configure.
foreach(_v HARNESS_DIR:tools/harness DOCTEST_DIR:third_party/doctest)
  string(REPLACE ":" ";" _p "${_v}")
  list(GET _p 0 _name)
  list(GET _p 1 _rel)
  if(NOT AWVA_${_name})
    set(AWVA_${_name} "${CMAKE_CURRENT_SOURCE_DIR}/../${_rel}")
  elseif(NOT EXISTS "${AWVA_${_name}}")
    message(FATAL_ERROR "AWVA_${_name}=${AWVA_${_name}} doesn't exist (a stale cache? configure the build directory again, or drop -DAWVA_${_name})")
  endif()
endforeach()
cmake_path(NORMAL_PATH AWVA_HARNESS_DIR)
cmake_path(NORMAL_PATH AWVA_DOCTEST_DIR)
set(_gen "${CMAKE_CURRENT_BINARY_DIR}/generated")

# --- the scripted performances (tools/harness/performance.h, perf_<name>.h) ----------------------------
function(awva_performance_source out name header)
  string(MAKE_C_IDENTIFIER "${name}" id)
  file(CONFIGURE OUTPUT "${out}" @ONLY CONTENT "// generated by sdk/CMakeLists.txt: the scripted performance \"${name}\" (${header})
#include <math.h>
#include <stdint.h>
#include <stdio.h>
#include <string.h>

#include \"awva_sdk/performance.h\"

namespace perf_${id} {
#include \"${header}\"
}

static void perf_block(int b, int nv, awva::dsp::VoiceControls* vc, awva::dsp::GlobalControls& g) {
	perf_${id}::Ctl v[NVOICES_MAX];
	float gl[8];
	perf_${id}::block_controls(b, v, nv, gl);
	for (int i = 0; i < nv; i++) {
		vc[i].note = v[i].note;
		vc[i].pres = v[i].pres;
		vc[i].vpres = v[i].vpres;
		vc[i].but_x = v[i].but_x;
		vc[i].but_y = v[i].but_y;
	}
	g = {gl[0], gl[1], gl[2], gl[3], gl[4], gl[5], gl[6], gl[7]};
}

extern const awva::sdk::ScriptedPerformance awva_perf_${id};
const awva::sdk::ScriptedPerformance awva_perf_${id} = {\"${name}\", \"${header}\", SAMPLERATE, NVOICES_MAX, perf_block};
")
endfunction()

file(GLOB _perf_headers "${AWVA_HARNESS_DIR}/perf_*.h")
list(SORT _perf_headers)
list(PREPEND _perf_headers "${AWVA_HARNESS_DIR}/performance.h")
set(_perf_srcs "")
set(_perf_decls "")
set(_perf_list "")
set(AWVA_PERFORMANCES "")
foreach(h IN LISTS _perf_headers)
  get_filename_component(hn "${h}" NAME_WE)
  if(hn STREQUAL "performance")
    set(name default)
  else()
    string(REGEX REPLACE "^perf_" "" name "${hn}")
  endif()
  string(MAKE_C_IDENTIFIER "${name}" id)
  awva_performance_source("${_gen}/perf_${id}.cpp" "${name}" "${h}")
  list(APPEND _perf_srcs "${_gen}/perf_${id}.cpp")
  list(APPEND AWVA_PERFORMANCES ${name})
  string(APPEND _perf_decls "extern const awva::sdk::ScriptedPerformance awva_perf_${id};\n")
  string(APPEND _perf_list "\t    &awva_perf_${id},\n")
endforeach()
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(AWVA_PERFORMANCES "${AWVA_PERFORMANCES}" PARENT_SCOPE) # (tests/CMakeLists.txt's render tests)
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${AWVA_HARNESS_DIR}")
file(CONFIGURE OUTPUT "${_gen}/performances.cpp" @ONLY CONTENT "// generated by sdk/CMakeLists.txt: the scripted performances of tools/harness
#include \"awva_sdk/performance.h\"

${_perf_decls}
std::span<const awva::sdk::ScriptedPerformance* const> awva::sdk::performances() {
	static const ScriptedPerformance* const list[] = {
${_perf_list}	};
	return list;
}
")
add_library(awva_performances STATIC ${_perf_srcs} "${_gen}/performances.cpp")
target_link_libraries(awva_performances PUBLIC awva_sdk_options)
target_compile_definitions(awva_performances PRIVATE _USE_MATH_DEFINES)

# --- render_custom: one engine and one performance with extra compiler flags ---------------------------
# (configured with -DAWVA_RENDER_VOICE=<engine> and the rest). The performance is compiled into the engine's
# translation unit, so its test hooks (BOWED_TEST_HOOK and the like) reach the engine.
set(AWVA_RENDER_VOICE "" CACHE STRING "render_custom: engine (empty: no render_custom)")
set(AWVA_RENDER_PERF "performance.h" CACHE STRING "render_custom: performance header")
set(AWVA_RENDER_PERF_DIR "" CACHE PATH "render_custom: extra directory searched for the performance header first")
set(AWVA_RENDER_FLAGS "" CACHE STRING "render_custom: extra compiler flags")
if(AWVA_RENDER_VOICE IN_LIST AWVA_ENGINES)
  separate_arguments(_custom_flags UNIX_COMMAND "${AWVA_RENDER_FLAGS}")
  set(v "${AWVA_RENDER_VOICE}")
  # the performance: in AWVA_RENDER_PERF_DIR if it is there, else in tools/harness (the log and
  # the WAV name the file used)
  set(_perf "${AWVA_HARNESS_DIR}/${AWVA_RENDER_PERF}")
  if(AWVA_RENDER_PERF_DIR AND EXISTS "${AWVA_RENDER_PERF_DIR}/${AWVA_RENDER_PERF}")
    set(_perf "${AWVA_RENDER_PERF_DIR}/${AWVA_RENDER_PERF}")
  elseif(IS_ABSOLUTE "${AWVA_RENDER_PERF}")
    set(_perf "${AWVA_RENDER_PERF}")
  endif()
  if(NOT EXISTS "${_perf}")
    message(FATAL_ERROR "render_custom: no performance ${AWVA_RENDER_PERF} (in ${AWVA_RENDER_PERF_DIR} or ${AWVA_HARNESS_DIR})")
  endif()
  file(CONFIGURE OUTPUT "${_gen}/custom.cpp" @ONLY CONTENT "// generated by sdk/CMakeLists.txt: render_custom's engine (${v}) and performance (${AWVA_RENDER_PERF})
#include \"${AWVA_ENGINE_${v}_TRAITS}\"
#include \"engine_entry.h\"
#include \"awva_sdk/performance.h\"
#include <math.h>
#include <stdint.h>
#include <stdio.h>
#include <string.h>
#include \"${_perf}\"

static void perf_block(int b, int nv, awva::dsp::VoiceControls* vc, awva::dsp::GlobalControls& g) {
	Ctl v[NVOICES_MAX];
	float gl[8];
	block_controls(b, v, nv, gl);
	for (int i = 0; i < nv; i++) {
		vc[i].note = v[i].note;
		vc[i].pres = v[i].pres;
		vc[i].vpres = v[i].vpres;
		vc[i].but_x = v[i].but_x;
		vc[i].but_y = v[i].but_y;
	}
	g = {gl[0], gl[1], gl[2], gl[3], gl[4], gl[5], gl[6], gl[7]};
}
static const awva::sdk::ScriptedPerformance perf = {\"custom\", \"${_perf}\", SAMPLERATE, NVOICES_MAX, perf_block};

std::span<const awva::sdk::EngineEntry* const> awva::sdk::engines() {
	static const EngineEntry* const list[] = {&entry_of<awva::dsp::engines::${v}>(\"${AWVA_ENGINE_${v}_HEADER}\")};
	return list;
}
std::span<const awva::sdk::ScriptedPerformance* const> awva::sdk::performances() {
	static const ScriptedPerformance* const list[] = {&perf};
	return list;
}
")
  add_executable(render_custom tools/awva_render.cpp "${_gen}/custom.cpp" src/registry.cpp)
  target_include_directories(render_custom PRIVATE "${AWVA_HARNESS_DIR}")
  target_link_libraries(render_custom PRIVATE awva_sdk)
  target_compile_definitions(render_custom PRIVATE _USE_MATH_DEFINES)
  target_compile_options(render_custom PRIVATE ${_custom_flags})
elseif(AWVA_RENDER_VOICE)
  message(WARNING "AWVA_RENDER_VOICE=${AWVA_RENDER_VOICE} is not an engine (${AWVA_ENGINES})")
endif()

# the renderer says what it played, and refuses what it can't find
set(_rec "${AWVA_HARNESS_DIR}/rec/play_2026-10-22_staccato.csv")
add_test(NAME awva_render.log_names_engine_preset_performance
  COMMAND awva_render --engine va --preset pad --seconds 0.5)
set_tests_properties(awva_render.log_names_engine_preset_performance PROPERTIES LABELS unit
  PASS_REGULAR_EXPRESSION "engine va \\(voice_va.h\\), preset [0-9]+ \"pad\".*input scripted performance \"default\" \\([^\n]*performance.h \\([0-9]+ bytes, fnv [0-9a-f]+\\)\\) \\[the default input")
if(EXISTS "${_rec}")
  add_test(NAME awva_render.log_names_recording COMMAND awva_render --engine ep --rec "${_rec}" --seconds 0.5)
  set_tests_properties(awva_render.log_names_recording PROPERTIES LABELS unit
    PASS_REGULAR_EXPRESSION "engine ep \\(voice_ep.h\\), preset \\(its default\\) 0 .*input key recording [^\n]*play_2026-10-22_staccato.csv \\([0-9]+ bytes, fnv [0-9a-f]+\\): [0-9]+ key events")
endif()
foreach(t IN ITEMS "unknown_engine;--engine nosuch" "unknown_preset;--engine va --preset nosuch"
                   "unknown_performance;--engine va --perf nosuch" "missing_recording;--engine va --rec /nonexistent.csv"
                   "missing_midi;--engine va --midi /nonexistent.mid" "two_inputs;--engine va --perf default --rec x.csv"
                   "too_many_voices;--engine trumpet --nv 64")
  list(GET t 0 name)
  list(GET t 1 args)
  separate_arguments(args)
  add_test(NAME awva_render.refuses_${name} COMMAND awva_render ${args})
  set_tests_properties(awva_render.refuses_${name} PROPERTIES LABELS unit WILL_FAIL TRUE)
endforeach()
