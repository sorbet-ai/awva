# The engine components (include/nsc/nsc.h, the C ABI; include/nsc/shim.h, any VoiceEngine as one):
#   nsc_<name>         each engine as its own shared library, <build>/components/<name>.dll /
#                      .dylib / .so, and its manifest <name>.json (id, name, summary, ABI, version,
#                      schema, sizes) written by nsc_manifest from the built library itself
#   nsc_manifest       loads a component and prints its manifest (also the loader the tests use)
#   awva_conformance   the ABI's conformance checks on any component library (conformance/)
#   tests: component.* every component loads, its manifest matches, and it renders bit for bit
#                      as the engine linked in (process_voices) and as the SDK's MIDI player (process)
#          conformance.<name>.*  awva_conformance on each engine's library
# Built with the SDK's flags (awva_sdk_options: -O2, no FMA contraction), so a component's samples
# are the SDK's, and an including project's when it is built with them.
set(_cgen "${CMAKE_CURRENT_BINARY_DIR}/generated")

# AWVA_VERSION: the version the components report (nsc_desc.version). Given (-DAWVA_VERSION=...),
# that; else git describe of this repository's own checkout (a submodule's too), never of an
# enclosing one (an export has no .git here: "unknown")
set(AWVA_VERSION "" CACHE STRING "the components' version (default: git describe of awva)")
set(_nsc_version "${AWVA_VERSION}")
if(NOT _nsc_version)
  set(_nsc_version "unknown")
  find_package(Git QUIET)
  if(GIT_FOUND AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../.git")
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${CMAKE_CURRENT_SOURCE_DIR}/.." describe --always --dirty
      OUTPUT_VARIABLE _nsc_version OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  endif()
  if(NOT _nsc_version) # (a .git that git can't read: a copied worktree, a container)
    set(_nsc_version "unknown")
  endif()
endif()

# awva_add_component(<target> <engine> STATIC|SHARED|MODULE): engine <engine> as a component
# library, for a project that builds its own host on the ABI (a plugin): the entry, the shim, the
# MIDI side and the declared defaults
function(awva_add_component target engine type)
  set(_g "${CMAKE_CURRENT_BINARY_DIR}/awva_components/${target}") # (the caller's build directory)
  file(MAKE_DIRECTORY "${_g}")
  add_custom_command(OUTPUT "${_g}/defaults_${engine}.h" COMMAND awva_defaults ${engine} "${_g}/defaults_${engine}.h"
    DEPENDS awva_defaults VERBATIM)
  file(CONFIGURE OUTPUT "${_g}/component.cpp" @ONLY CONTENT "// generated by awva_add_component: engine ${engine} as a component
#include \"${AWVA_ENGINE_${engine}_TRAITS}\"
#include \"nsc/shim.h\"
#include \"defaults_${engine}.h\"

NSC_DEFINE_ENTRY(awva::engines::${engine})
")
  add_library(${target} ${type} "${_g}/component.cpp" "${_g}/defaults_${engine}.h" "${AWVA_COMPONENT_SOURCE_DIR}/../sdk/src/midi_voices.cpp")
  target_link_libraries(${target} PRIVATE nsc_options)
  target_include_directories(${target} PUBLIC "${AWVA_COMPONENT_SOURCE_DIR}/include" PRIVATE "${_g}")
  set_target_properties(${target} PROPERTIES CXX_VISIBILITY_PRESET hidden POSITION_INDEPENDENT_CODE ON)
endfunction()
set(AWVA_COMPONENT_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}" CACHE INTERNAL "")

# --- engine packages (.awvapkg, docs/packages.md: the one file an engine ships in) -----------------
#   package_<id>   <AWVA_PACKAGE_DIR>/<id>-<slug>.awvapkg: the component's library as this PC's
#                  image, built with AWVA_PACKAGE_MIDI (below; its own target, package_lib_<id>)
#                  (tools/nsc_pack.py host, from the library and its nsc_manifest JSON) and the
#                  engine's preset files (presets/<id>), labelled AWVA_PACKAGE_LABEL (default: the
#                  id) and stamped with AWVA_PACKAGE_AWVA (default: this build's version, when it is
#                  a commit). Not built by default: cmake --build <dir> --target package_<id>.
#   packages       every engine's
# The image's tag is this PC's: x86_64-windows, arm64-macos, x86_64-macos, x86_64-linux, ...

set(AWVA_PACKAGE_SLUG "" CACHE STRING "the file name's part after the id (default: the awva commit)")
set(AWVA_PACKAGE_AWVA "" CACHE STRING "the awva commit packages say they are built on (default: this build's version)")
# A package is one file for every host, an embedded host's flash slots and a host's virtual
# device included, so its libraries are built with the MIDI tables those take (timed_midi.h,
# midi_voices.h: a block's events, the hosts' note ids, a step's messages); the descriptor says them
# (max_events, max_note_ids) to every host
set(AWVA_PACKAGE_MIDI AWVA_MIDI_QUEUE=256 AWVA_NOTE_IDS=16 AWVA_MIDI_STEP_QUEUE=256 CACHE STRING
  "the MIDI table sizes a package's libraries are built with")
if(WIN32)
  set(_pkg_os windows)
elseif(APPLE)
  set(_pkg_os macos)
else()
  set(_pkg_os linux)
endif()

  set(AWVA_PACKAGE_TAG "arm64-${_pkg_os}" CACHE INTERNAL "")
else()
  set(AWVA_PACKAGE_TAG "x86_64-${_pkg_os}" CACHE INTERNAL "")
endif()

set(_pkg_awva "${AWVA_PACKAGE_AWVA}")

  set(_pkg_awva "${_nsc_version}")
endif()

# awva_add_package(<target> <component target> <id> <out.awvapkg> [LABEL <text>]): a package of
# a component library built here (its presets: presets/<id>, when the folder is there). The
# library must be built with AWVA_PACKAGE_MIDI (awva_add_package_library makes one so): the
# packager refuses another.
function(awva_add_package target component id out)
  cmake_parse_arguments(PARSE_ARGV 4 A "" "LABEL" "")
  if(NOT Python3_Interpreter_FOUND)
    return()
  endif()
  set(_label "${A_LABEL}")
  if(NOT _label)
    set(_label "${AWVA_PACKAGE_LABEL}")
  endif()
  if(NOT _label)
    set(_label "${id}")
  endif()
  set(_awva "")
  if(_pkg_awva)
    set(_awva --awva "${_pkg_awva}")
  endif()
  set(_presets "")
  if(EXISTS "${AWVA_COMPONENT_SOURCE_DIR}/../presets/${id}")
    set(_presets --presets "${AWVA_COMPONENT_SOURCE_DIR}/../presets/${id}")
  endif()
  get_filename_component(_dir "${out}" DIRECTORY)
  add_custom_command(OUTPUT "${out}"
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${_dir}"
    COMMAND nsc_manifest "$<TARGET_FILE:${component}>" "${out}.json"
    COMMAND "${Python3_EXECUTABLE}" "${AWVA_PACK}" host "$<TARGET_FILE:${component}>" "${out}.json" ${AWVA_PACKAGE_TAG}
            "${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}" "${out}.nscm" --midi "${_pkg_midi}"
    COMMAND "${Python3_EXECUTABLE}" "${AWVA_PACK}" package "${out}" --image "${AWVA_PACKAGE_TAG}=${out}.nscm"
            --manifest "${out}.json" --label "${_label}" ${_awva} ${_presets}
    DEPENDS ${component} nsc_manifest "${AWVA_PACK}" VERBATIM)
  add_custom_target(${target} DEPENDS "${out}")
endfunction()
set(_all_packages "")
# a component library for packages: awva_add_component with AWVA_PACKAGE_MIDI
function(awva_add_package_library target engine)
  awva_add_component(${target} ${engine} SHARED)
  target_compile_definitions(${target} PRIVATE ${AWVA_PACKAGE_MIDI})
  set_target_properties(${target} PROPERTIES OUTPUT_NAME ${engine} PREFIX "" CXX_VISIBILITY_PRESET hidden
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/package_libs" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/package_libs")
  if(APPLE)
    target_link_options(${target} PRIVATE -Wl,-exported_symbol,_nsc_entry)
  endif()
  set_target_properties(${target} PROPERTIES EXCLUDE_FROM_ALL ON)
endfunction()

  endif()
endforeach()
