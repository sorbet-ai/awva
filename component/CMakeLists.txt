# The engine components (include/nsc/nsc.h, the C ABI; include/nsc/shim.h, any VoiceEngine as one):
#   nsc_<name>         each engine as its own shared library, <build>/components/<name>.dll /
#                      .dylib / .so, and its manifest <name>.json (id, name, summary, ABI, version,
#                      schema, sizes) written by nsc_manifest from the built library itself
#   nsc_manifest       loads a component and prints its manifest (also the loader the tests use)
#   tests: component.* every component loads, its manifest matches, and it renders bit for bit
#                      as the engine linked in (process_voices) and as the SDK's MIDI player (process)
# Built with the SDK's flags (awva_sdk_options: -O2, no FMA contraction), so a component's samples
# are the SDK's, and the virtual board's when it is built with them.
set(_cgen "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(AWVA_COMPONENT_DIR "${CMAKE_BINARY_DIR}/components" CACHE PATH "where the component libraries go")
find_package(Git QUIET)
set(_nsc_version "unknown")
if(GIT_FOUND)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${CMAKE_CURRENT_SOURCE_DIR}" describe --always --dirty
    OUTPUT_VARIABLE _nsc_version OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
endif()

add_library(nsc_options INTERFACE)
target_include_directories(nsc_options INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(nsc_options INTERFACE awva_sdk_options)
target_compile_definitions(nsc_options INTERFACE "NSC_VERSION=\"${_nsc_version}\"")

add_executable(nsc_manifest tools/nsc_manifest.cpp)
target_include_directories(nsc_manifest PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(nsc_manifest PRIVATE awva_host_options ${CMAKE_DL_LIBS})

# the parameters' declared defaults, measured from the engines linked in (component/tools/awva_defaults.cpp)
add_executable(awva_defaults tools/awva_defaults.cpp)
target_link_libraries(awva_defaults PRIVATE awva_sdk awva_engines)

set(AWVA_COMPONENTS "")
foreach(v IN LISTS AWVA_ENGINES)
  add_custom_command(OUTPUT "${_cgen}/defaults_${v}.h" COMMAND awva_defaults ${v} "${_cgen}/defaults_${v}.h"
    DEPENDS awva_defaults VERBATIM)
  file(CONFIGURE OUTPUT "${_cgen}/nsc_${v}.cpp" @ONLY CONTENT "// generated by component/CMakeLists.txt: engine ${v} as a component
#include \"${AWVA_ENGINE_${v}_TRAITS}\"
#include \"nsc/shim.h\"
#include \"defaults_${v}.h\"

NSC_DEFINE_ENTRY(awva::dsp::engines::${v})
")
  add_library(nsc_${v} SHARED "${_cgen}/nsc_${v}.cpp" "${_cgen}/defaults_${v}.h" "${CMAKE_CURRENT_SOURCE_DIR}/../sdk/src/midi_voices.cpp")
  target_link_libraries(nsc_${v} PRIVATE nsc_options)
  target_include_directories(nsc_${v} PRIVATE "${_cgen}")
  set_target_properties(nsc_${v} PROPERTIES OUTPUT_NAME ${v} PREFIX "" CXX_VISIBILITY_PRESET hidden
    LIBRARY_OUTPUT_DIRECTORY "${AWVA_COMPONENT_DIR}" RUNTIME_OUTPUT_DIRECTORY "${AWVA_COMPONENT_DIR}")
  if(APPLE)
    target_link_options(nsc_${v} PRIVATE -Wl,-exported_symbol,_nsc_entry)
  endif()
  add_custom_command(TARGET nsc_${v} POST_BUILD
    COMMAND nsc_manifest "$<TARGET_FILE:nsc_${v}>" "${AWVA_COMPONENT_DIR}/${v}.json" VERBATIM)
  list(APPEND AWVA_COMPONENTS nsc_${v})
endforeach()
add_custom_target(nsc_components DEPENDS ${AWVA_COMPONENTS})
set(AWVA_COMPONENTS "${AWVA_COMPONENTS}" PARENT_SCOPE)

if(EXISTS "${AWVA_DOCTEST_DIR}/doctest/doctest.h")
  add_executable(nsc_component_tests "${CMAKE_CURRENT_SOURCE_DIR}/../sdk/tests/main.cpp" tests/test_component.cpp)
  target_include_directories(nsc_component_tests PRIVATE "${AWVA_DOCTEST_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/include" "${AWVA_HARNESS_DIR}")
  target_link_libraries(nsc_component_tests PRIVATE awva_sdk awva_engines awva_performances ${CMAKE_DL_LIBS})
  target_compile_definitions(nsc_component_tests PRIVATE "NSC_COMPONENT_DIR=\"${AWVA_COMPONENT_DIR}\""
    "AWVA_REC_DIR=\"${AWVA_HARNESS_DIR}/rec\"")
  add_dependencies(nsc_component_tests nsc_components)
  doctest_discover_tests(nsc_component_tests TEST_PREFIX "component." PROPERTIES LABELS unit)
endif()
