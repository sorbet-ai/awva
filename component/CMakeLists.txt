# The engine components (include/nsc/nsc.h, the C ABI; include/nsc/shim.h, any VoiceEngine as one):
#   nsc_<name>         each engine as its own shared library, <build>/components/<name>.dll /
#                      .dylib / .so, and its manifest <name>.json (id, name, summary, ABI, version,
#                      schema, sizes) written by nsc_manifest from the built library itself
#   nsc_manifest       loads a component and prints its manifest (also the loader the tests use)
#   awva_conformance   the ABI's conformance checks on any component library (conformance/)
#   tests: component.* every component loads, its manifest matches, and it renders bit for bit
#                      as the engine linked in (process_voices) and as the SDK's MIDI player (process)
#          conformance.<name>.*  awva_conformance on each engine's library
# Built with the SDK's flags (awva_sdk_options: -O2, no FMA contraction), so a component's samples
# are the SDK's, and an including project's when it is built with them.
set(_cgen "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(AWVA_COMPONENT_DIR "${CMAKE_BINARY_DIR}/components" CACHE PATH "where the component libraries go")
# AWVA_VERSION: the version the components report (nsc_desc.version). Given (-DAWVA_VERSION=...),
# that; else git describe of this repository's own checkout (a submodule's too), never of an
# enclosing one (an export has no .git here: "unknown")
set(AWVA_VERSION "" CACHE STRING "the components' version (default: git describe of awva)")
set(_nsc_version "${AWVA_VERSION}")
if(NOT _nsc_version)
  set(_nsc_version "unknown")
  find_package(Git QUIET)
  if(GIT_FOUND AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../.git")
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${CMAKE_CURRENT_SOURCE_DIR}/.." describe --always --dirty
      OUTPUT_VARIABLE _nsc_version OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  endif()
  if(NOT _nsc_version) # (a .git that git can't read: a copied worktree, a container)
    set(_nsc_version "unknown")
  endif()
endif()

target_include_directories(nsc_options INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(nsc_options INTERFACE awva_sdk_options)
target_compile_definitions(nsc_options INTERFACE "NSC_VERSION=\"${_nsc_version}\"")

add_executable(nsc_manifest tools/nsc_manifest.cpp)
target_include_directories(nsc_manifest PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(nsc_manifest PRIVATE awva_host_options ${CMAKE_DL_LIBS})

# the parameters' declared defaults, measured from the engines linked in (component/tools/awva_defaults.cpp)

# awva_add_component(<target> <engine> STATIC|SHARED|MODULE): engine <engine> as a component
# library, for a project that builds its own host on the ABI (a plugin): the entry, the shim, the
# MIDI side and the declared defaults
function(awva_add_component target engine type)
  set(_g "${CMAKE_CURRENT_BINARY_DIR}/awva_components/${target}") # (the caller's build directory)
  file(MAKE_DIRECTORY "${_g}")
  add_custom_command(OUTPUT "${_g}/defaults_${engine}.h" COMMAND awva_defaults ${engine} "${_g}/defaults_${engine}.h"
    DEPENDS awva_defaults VERBATIM)
  file(CONFIGURE OUTPUT "${_g}/component.cpp" @ONLY CONTENT "// generated by awva_add_component: engine ${engine} as a component
#include \"${AWVA_ENGINE_${engine}_TRAITS}\"
#include \"nsc/shim.h\"
#include \"defaults_${engine}.h\"

NSC_DEFINE_ENTRY(awva::engines::${engine})
")
  add_library(${target} ${type} "${_g}/component.cpp" "${_g}/defaults_${engine}.h" "${AWVA_COMPONENT_SOURCE_DIR}/../sdk/src/midi_voices.cpp")
  target_link_libraries(${target} PRIVATE nsc_options)
  target_include_directories(${target} PUBLIC "${AWVA_COMPONENT_SOURCE_DIR}/include" PRIVATE "${_g}")
  set_target_properties(${target} PROPERTIES CXX_VISIBILITY_PRESET hidden POSITION_INDEPENDENT_CODE ON)
endfunction()
set(AWVA_COMPONENT_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}" CACHE INTERNAL "")

set(AWVA_COMPONENTS "")
foreach(v IN LISTS AWVA_ENGINES)
  awva_add_component(nsc_${v} ${v} SHARED)
  set_target_properties(nsc_${v} PROPERTIES OUTPUT_NAME ${v} PREFIX "" CXX_VISIBILITY_PRESET hidden
    LIBRARY_OUTPUT_DIRECTORY "${AWVA_COMPONENT_DIR}" RUNTIME_OUTPUT_DIRECTORY "${AWVA_COMPONENT_DIR}")
  if(APPLE)
    target_link_options(nsc_${v} PRIVATE -Wl,-exported_symbol,_nsc_entry)
  endif()
  add_custom_command(TARGET nsc_${v} POST_BUILD
    COMMAND nsc_manifest "$<TARGET_FILE:nsc_${v}>" "${AWVA_COMPONENT_DIR}/${v}.json" VERBATIM)
  list(APPEND AWVA_COMPONENTS nsc_${v})
endforeach()
# one component built with small MIDI tables, as a host with little RAM builds them (AWVA_MIDI_QUEUE,
# AWVA_NOTE_IDS, AWVA_MIDI_STEP_QUEUE: timed_midi.h, midi_voices.h): its descriptor says so and the
# conformance checks hold on it (conformance.sine_small.*)
awva_add_component(nsc_sine_small sine SHARED)
target_compile_definitions(nsc_sine_small PRIVATE AWVA_MIDI_QUEUE=256 AWVA_NOTE_IDS=16 AWVA_MIDI_STEP_QUEUE=256)
set_target_properties(nsc_sine_small PROPERTIES OUTPUT_NAME sine_small PREFIX "" CXX_VISIBILITY_PRESET hidden
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/small" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/small")
if(APPLE)
  target_link_options(nsc_sine_small PRIVATE -Wl,-exported_symbol,_nsc_entry)
endif()
add_custom_command(TARGET nsc_sine_small POST_BUILD
  COMMAND nsc_manifest "$<TARGET_FILE:nsc_sine_small>" "${CMAKE_CURRENT_BINARY_DIR}/small/sine_small.json" VERBATIM)
list(APPEND AWVA_COMPONENTS nsc_sine_small)
